1）postgresql.conf

2）pg_hba.conf
# 允许特定 IP 地址（如 192.168.1.101）通过密码认证连接
host    all             all             192.168.1.101/32        scram-sha-256

# 允许整个子网（如 192.168.1.0/24）通过密码认证连接

# 如果是公网环境，限制具体 IP（不建议直接开放 0.0.0.0/0）

1. jiebacfg：默认精确分词配置（最常用）
分词模式：基于 jieba 的「精确模式」，会将文本切分成最合理的词语组合（不重复、不冗余）。
适用场景：通用中文分词场景（如文章内容、评论、标题等），兼顾准确性和效率。
示例：
sql
SELECT to_tsvector('jiebacfg', '我爱中国北京天安门');
-- 结果：'中国':3 '北京':4 '天安门':5 '我爱':1,2

2. jiebaqry：查询专用分词配置
分词模式：基于 jieba 的「搜索引擎模式」，对长词进行二次切分，适合查询场景。
适用场景：用户输入的查询词（如搜索框关键词），确保更多相关词被匹配。
示例：
sql
SELECT to_tsvector('jiebaqry', '我爱中国北京天安门');
-- 结果可能包含更多细分词：'中国':3 '北京':4 '天安':5 '天安门':5 '我爱':1,2 '门':6


select to_tsvector('jiebacfg', '我爱中国北京天安门') @@ to_tsquery('jiebacfg', '美国') AS a;


四）安装vector
sudo dnf install -y postgresql17-devel gcc make

安装 Rust 工具链（vector 扩展使用 Rust 开发）：
curl --proto '=https' --tlsv1.2 -sSf https://sh.rustup.rs | sh
source $HOME/.cargo/env

git clone https://github.com/pgvector/pgvector.git
cd pgvector
make PG_CONFIG=/usr/pgsql-17/bin/pg_config
sudo make PG_CONFIG=/usr/pgsql-17/bin/pg_config install

# 进入 PostgreSQL
sudo -u postgres psql

# 在数据库中创建扩展
CREATE EXTENSION vector;

# 验证安装
SELECT * FROM pg_extension WHERE extname = 'vector';
